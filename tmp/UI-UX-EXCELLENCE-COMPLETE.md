## 📊 Before/After Comparison

### Modal Component Example

**Before:**
```tsx
// Limited configurability
<Modal isOpen={open} onClose={handleClose} title="Delete Item">
  <p>Are you sure?</p>
</Modal>
```

**After:**
```tsx
// Full control over behavior and size
<Modal
  isOpen={open}
  onClose={handleClose}
  title="Delete Item"
  size="sm"
  scrollBehavior="inside"
  disableOverlayClick
  disableEscapeKey
>
  <p>Are you sure?</p>
</Modal>
```

### Table Component Example

**Before:**
```tsx
// Basic unstyled table
<Table>
  <TableHeader>
    <TableRow>
      <TableHead>Name</TableHead>
    </TableRow>
  </TableHeader>
</Table>
```

**After:**
```tsx
// Styled, sortable, interactive table
<Table variant="striped" size="compact">
  <TableHeader>
    <TableRow>
      <TableHead
        sortable
        sortDirection="asc"
        onSort={() => handleSort('name')}
      >
        Name
      </TableHead>
    </TableRow>
  </TableHeader>
  <TableBody>
    <TableRow highlighted>
      <TableCell>Selected</TableCell>
    </TableRow>
  </TableBody>
</Table>
```

### PasswordInput Component Example

**Before:**
```tsx
// Basic password input with toggle only
<PasswordInput placeholder="Enter password" />
```

**After:**
```tsx
// Full strength validation with checklist
<PasswordInput
  placeholder="Enter password"
  showStrength
  showRequirements
  requirements={[
    { test: /.{8,}/, label: 'At least 8 characters' },
    { test: /[A-Z]/, label: 'One uppercase letter' },
    { test: /[0-9]/, label: 'One number' }
  ]}
/>
```
